cmake_minimum_required(VERSION 3.24)
project(vtc_truck_hub_plugin LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
add_library(vtc_truck_hub SHARED src/plugin.cpp)
target_compile_definitions(vtc_truck_hub PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN)
if(NOT DEFINED SCS_SDK_INCLUDE)
  message(FATAL_ERROR "SCS_SDK_INCLUDE muss auf den include-Ordner des offiziellen SCS Telemetry SDK zeigen")
endif()
target_include_directories(vtc_truck_hub PRIVATE ${SCS_SDK_INCLUDE})
target_link_libraries(vtc_truck_hub PRIVATE winhttp)
set_target_properties(vtc_truck_hub PROPERTIES OUTPUT_NAME "vtc_truck_hub")
if(WIN32)
  set_target_properties(vtc_truck_hub PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
endif()
